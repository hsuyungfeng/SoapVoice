```python
"""本機暫存管理模組"""
import json
import logging
from pathlib import Path
from typing import List, Optional, Dict, Any
from datetime import datetime
import hashlib

logger = logging.getLogger(__name__)

class PendingUpload:
    """
    暫存上傳紀錄（dataclass-like）
    """
    def __init__(
        self,
        session_id: str,
        audio_path: str,
        screenshot_path: str,
        patient_name: str = "",
        doctor_name: str = "",
        queued_at: str = None,
        retry_count: int = 0
    ):
        self.session_id = session_id
        self.audio_path = audio_path
        self.screenshot_path = screenshot_path
        self.patient_name = patient_name
        self.doctor_name = doctor_name
        self.queued_at = queued_at or datetime.now().isoformat()
        self.retry_count = retry_count
    
    def to_dict(self) -> Dict[str, Any]:
        """序列化為 dict"""
        return {
            "session_id": self.session_id,
            "audio_path": self.audio_path,
            "screenshot_path": self.screenshot_path,
            "patient_name": self.patient_name,
            "doctor_name": self.doctor_name,
            "queued_at": self.queued_at,
            "retry_count": self.retry_count
        }
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "PendingUpload":
        """從 dict 反序列化"""
        return cls(**data)

class LocalStorageManager:
    """
    本機暫存管理器
    - 管理待上傳隊列
    - 持久化為 JSON 檔案
    - 支援載入、新增、移除、查詢
    """
    
    # 暫存目錄結構
    # ~/AppData/Local/SoapVoice/pending/
    #   ├── queue.json（序列化隊列）
    #   ├── pending_20260427_103000_a1b2c3d4.json（個別紀錄）
    #   └── ...
    
    def __init__(self, storage_dir: str):
        self.storage_dir = Path(storage_dir)
        self.storage_dir.mkdir(parents=True, exist_ok=True)
        
        self.pending_dir = self.storage_dir / "pending"
        self.pending_dir.mkdir(exist_ok=True)
        
        self.queue_file = self.pending_dir / "queue.json"
        self.pending_uploads: List[PendingUpload] = []
        
        # 啟動時載入既有隊列
        self._load_queue()
    
    def _load_queue(self):
        """從 queue.json 載入既有隊列"""
        if self.queue_file.exists():
            try:
                with open(self.queue_file, "r", encoding="utf-8") as f:
                    data = json.load(f)
                self.pending_uploads = [PendingUpload.from_dict(item) for item in data]
                logger.info(f"Loaded {len(self.pending_uploads)} pending uploads from queue")
            except Exception as e:
                logger.error(f"Failed to load queue: {e}")
                self.pending_uploads = []
        else:
            self.pending_uploads = []
    
    def _save_queue(self):
        """將隊列序列化到 queue.json"""
        try:
            with open(self.queue_file, "w", encoding="utf-8") as f:
                data = [p.to_dict() for p in self.pending_uploads]
                json.dump(data, f, ensure_ascii=False, indent=2)
            logger.info(f"Queue saved ({len(self.pending_uploads)} items)")
        except Exception as e:
            logger.error(f"Failed to save queue: {e}")
    
    def add_pending(
        self,
        session_id: str,
        audio_path: str,
        screenshot_path: str,
        patient_name: str = "",
        doctor_name: str = ""
    ) -> bool:
        """
        新增待上傳項目
        傳回: True 成功，False 失敗或隊列滿
        """
        # 檢查隊列上限
        if len(self.pending_uploads) >= 10:
            logger.warning("Pending queue full, removing oldest item")
            self._remove_oldest()
        
        # 驗證音檔存在
        if not Path(audio_path).exists():
            logger.error(f"Audio file not found: {audio_path}")
            return False
        
        # 新增項目
        pending = PendingUpload(
            session_id=session_id,
            audio_path=audio_path,
            screenshot_path=screenshot_path,
            patient_name=patient_name,
            doctor_name=doctor_name
        )
        
        self.pending_uploads.append(pending)
        self._save_queue()
        
        logger.info(f"Added pending upload: {session_id}")
        return True
    
    def get_all_pending(self) -> List[PendingUpload]:
        """取得所有待上傳項目"""
        return self.pending_uploads.copy()
    
    def get_next_pending(self) -> Optional[PendingUpload]:
        """取得隊列中的第一項（待重試）"""
        return self.pending_uploads[0] if self.pending_uploads else None
    
    def remove_pending(self, session_id: str) -> bool:
        """
        移除特定項目（成功上傳後呼叫）
        傳回: True 成功，False 項目不存在
        """
        original_count = len(self.pending_uploads)
        self.pending_uploads = [p for p in self.pending_uploads if p.session_id != session_id]
        
        if len(self.pending_uploads) < original_count:
            self._save_queue()
            logger.info(f"Removed pending upload: {session_id}")
            return True
        
        return False
    
    def increment_retry_count(self, session_id: str) -> bool:
        """
        增加重試計數
        傳回: True 成功，False 項目不存在
        """
        for pending in self.pending_uploads:
            if pending.session_id == session_id:
                pending.retry_count += 1
                self._save_queue()
                return True
        return False
    
    def _remove_oldest(self):
        """移除最舊的項目（隊列滿時）"""
        if self.pending_uploads:
            oldest = self.pending_uploads.pop(0)
            logger.warning(f"Removed oldest pending upload due to queue overflow: {oldest.session_id}")
            self._save_queue()
    
    def clear_all(self):
        """清空所有待上傳項目（謹慎使用）"""
        self.pending_uploads.clear()
        self._save_queue()
        logger.warning("Cleared all pending uploads")
    
    def get_queue_size(self) -> int:
        """取得隊列大小"""
        return len(self.pending_uploads)
```

"""重試管理與指數退避邏輯"""
import logging
import asyncio
from typing import Callable, Optional, Tuple
from datetime import datetime, timedelta
import math

logger = logging.getLogger(__name__)

class RetryManager:
    """
    重試管理器
    - 指數退避：2 → 4 → 8 秒
    - 最多 3 次重試
    - 追蹤重試狀態
    """
    
    def __init__(
        self,
        base_delay_sec: float = 2.0,
        max_retries: int = 3,
        backoff_multiplier: float = 2.0
    ):
        self.base_delay_sec = base_delay_sec
        self.max_retries = max_retries
        self.backoff_multiplier = backoff_multiplier
        
        # 重試狀態追蹤
        self.retry_state = {}  # session_id -> {"count": int, "next_retry_at": datetime}
    
    def get_retry_delay(self, retry_count: int) -> float:
        """
        計算指數退避延遲時間（秒）
        retry_count: 0 → base_delay
                     1 → base_delay * 2
                     2 → base_delay * 4
        """
        if retry_count < 0:
            return 0
        return self.base_delay_sec * math.pow(self.backoff_multiplier, retry_count)
    
    def should_retry(self, session_id: str, current_retry_count: int) -> bool:
        """
        檢查是否應進行重試
        傳回: True 應重試，False 已達最大次數
        """
        return current_retry_count < self.max_retries
    
    def get_next_retry_delay(self, current_retry_count: int) -> float:
        """
        取得下次重試的延遲時間
        """
        if not self.should_retry("", current_retry_count):
            return -1  # 不應重試
        return self.get_retry_delay(current_retry_count)
    
    def mark_retry_scheduled(self, session_id: str, retry_count: int):
        """標記重試已排程（用於追蹤）"""
        next_delay = self.get_retry_delay(retry_count)
        self.retry_state[session_id] = {
            "count": retry_count,
            "next_retry_at": datetime.now() + timedelta(seconds=next_delay),
            "last_attempt_at": datetime.now()
        }
    
    async def wait_before_retry(self, retry_count: int) -> bool:
        """
        等待指數退避延遲後再進行重試
        傳回: True 應進行重試，False 已達最大次數
        """
        if not self.should_retry("", retry_count):
            logger.info(f"Max retries ({self.max_retries}) reached, aborting")
            return False
        
        delay = self.get_retry_delay(retry_count)
        logger.info(f"Waiting {delay:.0f}s before retry #{retry_count + 1}")
        await asyncio.sleep(delay)
        return True
    
    def reset_state(self, session_id: str):
        """重置特定 session 的重試狀態"""
        self.retry_state.pop(session_id, None)
    
    def get_state(self, session_id: str) -> Optional[dict]:
        """取得重試狀態"""
        return self.retry_state.get(session_id)
```

    async def retry_pending_uploads(self):
        """連線恢復後自動重試本機暫存隊列"""
        while True:
            pending = self.storage.get_next_pending()
            if not pending:
                logger.info("No more pending uploads")
                break
            
            # 檢查是否應重試
            if not self.retry_manager.should_retry(pending.session_id, pending.retry_count):
                logger.error(f"Max retries reached for {pending.session_id}, removing from queue")
                self.storage.remove_pending(pending.session_id)
                continue
            
            # 等待指數退避延遲
            if pending.retry_count > 0:
                should_continue = await self.retry_manager.wait_before_retry(pending.retry_count)
                if not should_continue:
                    break
            
            # 嘗試重試
            logger.info(f"Retrying {pending.session_id} (attempt {pending.retry_count + 1})")
            result = await self._do_upload(
                audio_path=pending.audio_path,
                screenshot_path=pending.screenshot_path,
                patient_name=pending.patient_name,
                doctor_name=pending.doctor_name,
                session_id=pending.session_id
            )
            
            if result:
                # 成功，刪除本機暫存
                self.storage.remove_pending(pending.session_id)
                self.retry_manager.reset_state(pending.session_id)
                logger.info(f"Pending upload succeeded: {pending.session_id}")
                if self.result_callback:
                    await self.result_callback(result)
            else:
                # 失敗，增加重試計數並標記下次重試時間
                self.storage.increment_retry_count(pending.session_id)
                self.retry_manager.mark_retry_scheduled(pending.session_id, pending.retry_count + 1)
                logger.warning(f"Retry failed, will retry later")
                break  # 等待下次健康檢查
```
